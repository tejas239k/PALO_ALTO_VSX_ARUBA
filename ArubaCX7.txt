configure terminal

    ! 1. Create VLANs 10 and 20
    vlan 10
        name Client-VLAN-10
    exit

    vlan 20
        name Client-VLAN-20
    exit

    ! 2. Configure the ISL (Inter-Switch Link) LAG using ports 1/1/3 and 1/1/4
    interface lag 128
        description VSX-ISL
        no shutdown
        no routing
        vlan trunk allowed all
        lacp mode active
    exit

    interface 1/1/3-1/1/4
        no shutdown
        lag 128
    exit

    ! 3. Configure the VSX Keepalive Link
    interface 1/1/6
        description VSX-Keepalive
        no shutdown
        ip address 192.168.255.2/30
    exit

    ! 4. Enable VSX Global Settings (Secondary Role — No vsx-sync here)
    vsx
        inter-switch-link lag 128
        role secondary
        keepalive peer 192.168.255.1 source 192.168.255.2
    exit

    ! 5. Configure MC-LAG Bundle for PaloAlto1 (Port 1/1/1)
    interface lag 1 multi-chassis
        description MC-LAG-to-PaloAlto1-ae1
        no shutdown
        no routing
        vlan trunk allowed all
        lacp mode active
    exit

    interface 1/1/1
        description Physical-to-PaloAlto1-Eth1/5
        no shutdown
        lag 1
    exit

    ! 5.1 Configure MC-LAG Bundle for PaloAlto2 (Port 1/1/2)
    interface lag 2 multi-chassis
        description MC-LAG-to-PaloAlto2-ae1
        no shutdown
        no routing
        vlan trunk allowed all
        lacp mode active
    exit

    interface 1/1/2
        description Physical-to-PaloAlto2-Eth1/5
        no shutdown
        lag 2
    exit

    ! 6. Configure MC-LAG for the Patch Panel / Client Access (Port 1/1/5)
    interface lag 3 multi-chassis
        description MC-LAG-to-Patch-Panel-Trunk
        no shutdown
        no routing
        vlan trunk allowed 10,20
        lacp mode active
    exit

    interface 1/1/5
        no shutdown
        lag 3
    exit

end
